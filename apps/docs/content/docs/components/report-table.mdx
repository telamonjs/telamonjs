import { TypeTable } from 'fumadocs-ui/components/type-table'

A plain, read-only table for report sections and dashboard cards: it renders the first
`pageSize` rows, a "load more" button for the rest and a "Showing X of Y records" line. It
has no sorting, filtering or row actions — for CRUD lists use [ListPage](/docs/components/list-page)
or [DataTable](/docs/components/data-table). Also exported as `SimpleTable`.

---

## Basic Usage

```tsx
'use client'

import { ReportTable } from '@telamonjs/ui'

const columns = [
  { key: 'month', header: 'Month' },
  { key: 'customer.name', header: 'Customer' }, // dot paths read nested values
  {
    key: 'total',
    header: 'Total',
    className: 'text-right',
    render: (row: Revenue) => row.total.toFixed(2),
  },
]

export function RevenueTable({ rows }: { rows: Revenue[] }) {
  return <ReportTable data={rows} columns={columns} pageSize={20} />
}
```

## Examples

### Without the Footer

```tsx
<ReportTable data={topProducts} columns={columns} pageSize={10} showFooter={false} />
```

Hide the count line when the surrounding card already states the total, or when the table
is short enough that the count adds nothing.

### Keyboard-Scrollable Region

```tsx
<ReportTable
  data={rows}
  columns={columns}
  maxHeight='300px'
  scrollProps={{ tabIndex: 0, role: 'region', 'aria-label': 'Revenue by month' }}
/>
```
