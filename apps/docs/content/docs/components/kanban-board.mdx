## Basic Usage

```tsx
'use client'

import { KanbanBoard } from '@telamonjs/ui'

const columns = [
  {
    id: 'todo',
    title: 'To Do',
    cards: [
      { id: '1', title: 'Design homepage', description: 'Create mockups' },
      { id: '2', title: 'Setup database', description: 'PostgreSQL config' },
    ],
  },
  {
    id: 'in_progress',
    title: 'In Progress',
    cards: [
      { id: '3', title: 'Implement auth', description: 'NextAuth setup' },
    ],
  },
  {
    id: 'done',
    title: 'Done',
    cards: [
      { id: '4', title: 'Project setup', description: 'Next.js initialized' },
    ],
  },
]

export default function ProjectBoard() {
  function handleCardMove(
    cardId: string,
    fromColumn: string,
    toColumn: string
  ) {
    console.log(`Moved ${cardId} from ${fromColumn} to ${toColumn}`)
    // Update backend
  }

  return <KanbanBoard columns={columns} onCardMove={handleCardMove} />
}
```

## Complete Example

```tsx title="app/projects/[id]/board/page.tsx"
'use client'

import { useState } from 'react'
import { KanbanBoard, type KanbanColumn } from '@telamonjs/ui'
import { useMutation } from '@tanstack/react-query'
import { tasksApi } from '@/lib/api/tasks-client'

export default function ProjectBoard({
  initialData,
}: {
  initialData: KanbanColumn[]
}) {
  const [columns, setColumns] = useState(initialData)

  const moveMutation = useMutation({
    mutationFn: ({ cardId, toColumn }: { cardId: string; toColumn: string }) =>
      tasksApi.updateStatus(cardId, toColumn),
  })

  function handleCardMove(cardId: string, fromColumn: string, toColumn: string) {
    // Optimistic update
    setColumns(cols => {
      const card = cols
        .find(c => c.id === fromColumn)
        ?.cards.find(c => c.id === cardId)
      if (!card) return cols
      return cols.map(col => {
        if (col.id === fromColumn)
          return { ...col, cards: col.cards.filter(c => c.id !== cardId) }
        if (col.id === toColumn) return { ...col, cards: [...col.cards, card] }
        return col
      })
    })

    // Persist to backend
    moveMutation.mutate({ cardId, toColumn })
  }

  return (
    <div className='container py-6'>
      <div className='mb-6'>
        <h1 className='text-3xl font-bold'>Project Board</h1>
        <p className='text-muted-foreground'>
          Drag and drop tasks to update their status
        </p>
      </div>

      <KanbanBoard
        columns={columns}
        onCardMove={handleCardMove}
        onCardClick={card => console.log('open task', card.id)}
      />
    </div>
  )
}
```

## State Management

**Client-side state:**

```tsx
const [columns, setColumns] = useState(initialData)
```

**With optimistic updates:**

```tsx
async function handleCardMove(cardId, from, to) {
  // 1. Update UI immediately
  setColumns(optimisticUpdate)

  // 2. Persist to backend
  try {
    await updateCard(cardId, { status: to })
  } catch (error) {
    // 3. Rollback on error
    setColumns(previousState)
    toast.error('Failed to update task')
  }
}
```

**With React Query (standard):**

```tsx
const moveMutation = useMutation({
  mutationFn: ({ cardId, toColumn }: { cardId: string; toColumn: string }) =>
    tasksApi.updateStatus(cardId, toColumn),
  onError: () => {
    setColumns(previousState)
    toast.error('Failed to update task')
  },
})
```

## Best Practices

1. **Limit columns** - 3-6 columns for optimal UX
2. **Set WIP limits** - Prevent bottlenecks (e.g., max 5 in "In Progress")
3. **Add filters** - Filter by assignee, priority, tags
4. **Optimistic updates** - Move the card in state immediately, persist async
5. **Mobile-friendly** - Pointer sensor covers touch drag

---

## Accessibility

- Drag-and-drop uses dnd-kit's `PointerSensor` (mouse + touch) with an
  activation distance so plain clicks still fire `onCardClick`
- Keyboard-driven dragging is **not** wired up (no `KeyboardSensor`) — if
  card movement must be keyboard accessible, offer an alternative action
  (e.g. a "move to column" menu via `onCardClick`)

---

## Performance

**Optimization tips:**

- Lazy load card details (fetch them in `onCardClick`)
- Debounce backend updates
- Use optimistic updates
- Every card renders (no virtualization) — filter or paginate very large columns before passing them
